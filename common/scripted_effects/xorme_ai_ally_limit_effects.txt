# =====================================================================
# AI ALLY LIMIT (max 2 AI allies)
# ---------------------------------------------------------------------
# Rule: an AI country may keep at most 2 allies that are themselves AI.
# Player allies are NEVER counted towards the limit and are NEVER broken,
# so an AI can still end up with (e.g.) 2 AI allies + 2 player allies.
#
# How it works:
#  - We only run on AI countries (the triggering country must have ai = yes).
#  - We iterate over every ally. Player allies are simply skipped.
#  - The first 2 AI allies are "kept" (a counter is decremented for each).
#  - Every further AI ally has its alliance with us dissolved.
# =====================================================================
xorme_ai_enforce_ally_limit_effect = {
	if = {
		limit = {
			ai = yes
			num_of_allies = 2
		}
		set_variable = {
			which = xorme_ai_allies_allowed
			value = 2
		}
		every_ally = {
			if = {
				limit = {
					ai = yes
				}
				if = {
					limit = {
						ROOT = {
							check_variable = {
								which = xorme_ai_allies_allowed
								value = 1
							}
						}
					}
					ROOT = {
						subtract_variable = {
							which = xorme_ai_allies_allowed
							value = 1
						}
					}
				}
				else = {
					# This AI ally is beyond the limit of 2 -> dissolve it.
					break_alliance = ROOT
				}
			}
		}
	}
}

xorme_ai_enforce_ally_opinion_gate_effect = {
	if = {
		limit = {
			ai = yes
			num_of_allies = 2
		}
		every_ally = {
			limit = {
				ROOT = { NOT = { has_opinion = { who = PREV value = 25 } } }
			}
			break_alliance = ROOT
		}
	}
}
