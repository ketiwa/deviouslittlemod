namespace = dev_cost_edict_events

# Hidden fallback event: applies the Dev Cost Edict modifier to countries that don't have it yet.
country_event = {
	id = dev_cost_edict_events.1
	title = dev_cost_edict_events.1.t
	desc = dev_cost_edict_events.1.desc
	picture = BATTLE_eventPicture

	trigger = {
		NOT = { has_country_modifier = dev_cost_edict_modifier }
	}

	mean_time_to_happen = {
		days = 1
	}

	hidden = yes

	immediate = {
		add_country_modifier = {
			name = "dev_cost_edict_modifier"
			duration = -1
			hidden = yes
		}
	}

	option = {
		name = dev_cost_edict_events.1.a
		ai_chance = { factor = 1 }
	}
}
